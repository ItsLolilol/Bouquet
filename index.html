      function normalize(v) {

        const len = Math.hypot(v[0], v[1], v[2]) || 1;

        return [v[0] / len, v[1] / len, v[2] / len];

      }


      function cross(a, b) {

        return [

          a[1] * b[2] - a[2] * b[1],

          a[2] * b[0] - a[0] * b[2],

          a[0] * b[1] - a[1] * b[0]

        ];

      }


      function basisFromDir(dir) {

        const up = normalize(dir);

        const helper = Math.abs(up[1]) < 0.9 ? [0, 1, 0] : [1, 0, 0];

        const right = normalize(cross(helper, up));

        const forward = normalize(cross(up, right));

        return { right, up, forward };

      }

      function addPoint(x, y, z, nx, ny, nz, color, charOverride = null) {

        points.push({

          x, y, z,

          nx, ny, nz,

          g: color[1],

          b: color[2],

          ch: charOverride

        });

      }


      // Fleur type cosmos / marguerite / tournesol

      function addCosmos(cx, cy, cz, dir, petalCount, radius, petalColor, centerColor) {

        const { right, up, forward } = basisFromDir(dir);


        // Cœur bombé de la fleur

        const centerR = radius * 0.30;

        for (let i = 0; i < 65; i++) {

          const u = Math.sqrt(i / 65);

          const theta = i * 2.39996; // Angle d'or

          const rr = u * centerR;

          const lx = Math.cos(theta) * rr;

          const lz = Math.sin(theta) * rr;

          const ly = Math.sqrt(Math.max(0, 1 - u * u)) * centerR * 0.75;


          const px = cx + right[0] * lx + up[0] * ly + forward[0] * lz;

          const py = cy + right[1] * lx + up[1] * ly + forward[1] * lz;

          const pz = cz + right[2] * lx + up[2] * ly + forward[2] * lz;


          const n = normalize([

            right[0] * lx + up[0] * (ly + 0.2) + forward[0] * lz,

            right[1] * lx + up[1] * (ly + 0.2) + forward[1] * lz,

            right[2] * lx + up[2] * (ly + 0.2) + forward[2] * lz

          ]);


          addPoint(px, py, pz, n[0], n[1], n[2], centerColor);

        }


        // Pétales en 3D légèrement recourbés

        for (let p = 0; p < petalCount; p++) {

          const baseAngle = (p / petalCount) * Math.PI * 2;

          const petalWidth = (Math.PI / petalCount) * 0.85;


          for (let ri = 1; ri <= 8; ri++) {

            const t = ri / 8;

            const dist = centerR * 0.85 + t * (radius - centerR * 0.85);

            const widthFactor = Math.sin(t * Math.PI) * petalWidth;

            const stepsW = Math.max(2, Math.round(4 * Math.sin(t * Math.PI)));


            for (let wi = -stepsW; wi <= stepsW; wi++) {

              const a = baseAngle + (wi / (stepsW || 1)) * widthFactor;

              const cup = Math.sin(t * Math.PI * 0.85) * radius * 0.22 - t * t * radius * 0.12;


              const lx = Math.cos(a) * dist;

              const lz = Math.sin(a) * dist;

              const ly = cup;


              const px = cx + right[0] * lx + up[0] * ly + forward[0] * lz;

              const py = cy + right[1] * lx + up[1] * ly + forward[1] * lz;

              const pz = cz + right[2] * lx + up[2] * ly + forward[2] * lz;


              const n = normalize([

                up[0] * 0.85 + right[0] * Math.cos(a) * 0.25 + forward[0] * Math.sin(a) * 0.25,

                up[1] * 0.85 + right[1] * Math.cos(a) * 0.25 + forward[1] * Math.sin(a) * 0.25,

                up[2] * 0.85 + right[2] * Math.cos(a) * 0.25 + forward[2] * Math.sin(a) * 0.25

              ]);


              addPoint(px, py, pz, n[0], n[1], n[2], petalColor);

            }

          }

        }

      }


      // Tulipe en 3D (forme de calice)

      function addTulip(cx, cy, cz, dir, scale, color) {

        const { right, up, forward } = basisFromDir(dir);

        const h = 0.62 * scale;

        const maxR = 0.34 * scale;


        for (let vi = 0; vi <= 14; vi++) {

          const v = vi / 14;

          const r = maxR * Math.pow(Math.sin(v * Math.PI * 0.82), 0.72);

          const y = (v - 0.1) * h;

          const ringSteps = Math.max(8, Math.round(24 * (r / maxR)));


          for (let ui = 0; ui < ringSteps; ui++) {

            const u = (ui / ringSteps) * Math.PI * 2;

            const lobe = 1 + 0.08 * Math.cos(u * 6) * v;

            const lx = Math.cos(u) * r * lobe;

            const lz = Math.sin(u) * r * lobe;


            const px = cx + right[0] * lx + up[0] * y + forward[0] * lz;

            const py = cy + right[1] * lx + up[1] * y + forward[1] * lz;

            const pz = cz + right[2] * lx + up[2] * y + forward[2] * lz;


            const n = normalize([

              right[0] * Math.cos(u) + forward[0] * Math.sin(u) + up[0] * 0.25,

              right[1] * Math.cos(u) + forward[1] * Math.sin(u) + up[1] * 0.25,

              right[2] * Math.cos(u) + forward[2] * Math.sin(u) + up[2] * 0.25

            ]);


            addPoint(px, py, pz, n[0], n[1], n[2], color);

          }

        }

      }


      // Tige courbée reliant la fleur au bas du bouquet

      function addStem(x0, y0, z0, x1, y1, z1) {

        const steps = 26;

        const midX = (x0 + x1) * 0.35;

        const midY = (y0 + y1) * 0.5;

        const midZ = (z0 + z1) * 0.35;


        for (let i = 0; i <= steps; i++) {

          const t = i / steps;

          const inv = 1 - t;

          const x = inv * inv * x0 + 2 * inv * t * midX + t * t * x1;

          const y = inv * inv * y0 + 2 * inv * t * midY + t * t * y1;

          const z = inv * inv * z0 + 2 * inv * t * midZ + t * t * z1;


          for (let a = 0; a < 4; a++) {

            const ang = (a / 4) * Math.PI * 2;

            const rad = 0.032;

            addPoint(

              x + Math.cos(ang) * rad,

              y,

              z + Math.sin(ang) * rad,

              Math.cos(ang), 0.15, Math.sin(ang),

              COLORS.stemGreen

            );

          }

        }

      }


      // Feuille verte allongée en 3D

      function addLeaf(ox, oy, oz, dir, length, width, color) {

        const { right, up, forward } = basisFromDir(dir);

        for (let i = 1; i <= 12; i++) {

          const t = i / 12;

          const w = Math.sin(t * Math.PI) * width;

          const droop = -t * t * length * 0.28;

          const wSteps = Math.max(1, Math.round(3 * Math.sin(t * Math.PI)));


          for (let j = -wSteps; j <= wSteps; j++) {

            const s = j / (wSteps || 1);

            const lx = s * w;

            const ly = t * length;

            const lz = droop + Math.abs(s) * w * 0.18;


            const px = ox + right[0] * lx + up[0] * ly + forward[0] * lz;

            const py = oy + right[1] * lx + up[1] * ly + forward[1] * lz;

            const pz = oz + right[2] * lx + up[2] * ly + forward[2] * lz;


            addPoint(px, py, pz, forward[0], forward[1] + 0.4, forward[2], color);

          }

        }

      }


      // Papier d'emballage conique + rabat arrière

      function addWrap() {

        // Cône principal (resserré à la taille y = -0.85, évasé vers le haut et le bas)

        const yTop = 0.45;

        const yWaist = -0.88;

        const yBot = -1.62;


        for (let i = 0; i <= 34; i++) {

          const t = i / 34;

          const y = yBot + t * (yTop - yBot);


          let radius;

          if (y >= yWaist) {

            const k = (y - yWaist) / (yTop - yWaist);

            radius = 0.24 + Math.pow(k, 1.15) * 0.88;

          } else {

            const k = (yWaist - y) / (yWaist - yBot);

            radius = 0.24 + k * 0.28;

          }


          const ringSteps = Math.max(18, Math.round(radius * 46));

          for (let j = 0; j < ringSteps; j++) {

            const a = (j / ringSteps) * Math.PI * 2;

            // Effet papier plié sur le bord supérieur

            const fold = y > -0.2 ? Math.sin(a * 5) * 0.045 * ((y + 0.2) / 0.65) : 0;

            // Ouverture légèrement plus basse devant pour laisser voir les fleurs

            if (y > 0.15 && Math.sin(a) > 0.25) continue;


            const r = radius + fold;

            const x = Math.cos(a) * r;

            const z = Math.sin(a) * r;


            addPoint(

              x, y, z,

              Math.cos(a), 0.22, Math.sin(a),

              Math.sin(a) < 0 ? COLORS.wrapBack : COLORS.wrapLight

            );

          }

        }


        // Grand papier plié qui remonte derrière le bouquet

        for (let i = 0; i <= 18; i++) {

          const t = i / 18;

          const y = 0.35 + t * 0.95;

          const span = 0.95 * (1 - t * 0.25);

          const steps = 22;


          for (let j = 0; j <= steps; j++) {

            const u = j / steps;

            const a = Math.PI + (u - 0.5) * 2.1; // Face arrière

            const r = 1.08 + t * 0.22 + Math.sin(u * Math.PI * 3) * 0.04;

            const x = Math.cos(a) * r * (span / 0.95);

            const z = Math.sin(a) * r;

            addPoint(x, y, z, Math.cos(a), 0.15, Math.sin(a), COLORS.wrapBack);

          }

        }

      }


      // Ruban et nœud autour de la taille du bouquet

      function addRibbon() {

        const yWaist = -0.88;

        const rWaist = 0.28;


        // Anneau du ruban

        for (let iy = -2; iy <= 2; iy++) {

          const y = yWaist + iy * 0.035;

          for (let j = 0; j < 28; j++) {

            const a = (j / 28) * Math.PI * 2;

            addPoint(

              Math.cos(a) * rWaist, y, Math.sin(a) * rWaist,

              Math.cos(a), 0, Math.sin(a),

              COLORS.ribbon

            );

          }

        }


        // Deux boucles du nœud devant (z > 0)

        for (const side of [-1, 1]) {

          for (let i = 0; i < 24; i++) {

            const t = (i / 24) * Math.PI * 2;

            const lx = side * (0.20 + Math.cos(t) * 0.18);

            const ly = yWaist + Math.sin(t) * 0.12;

            const lz = rWaist + 0.08 + Math.sin(t * 2) * 0.04;

            addPoint(lx, ly, lz, side * 0.3, Math.sin(t) * 0.4, 0.9, COLORS.ribbon);

          }

          // Rubans qui pendent

          for (let i = 0; i <= 14; i++) {

            const t = i / 14;

            const lx = side * (0.06 + t * 0.22);

            const ly = yWaist - 0.05 - t * 0.52;

            const lz = rWaist + 0.08 + t * 0.10;

            addPoint(lx, ly, lz, 0, 0.1, 1, COLORS.ribbon);

          }

        }

      }


      // Assemblage complet du bouquet

      function buildBouquet() {

        // Fleurs principales (cosmos, marguerites, tournesol, tulipes)

        const cosmosSpecs = [

          // [x, y, z, dir, petals, radius, petalColor, centerColor]

          [ 0.00,  1.28,  0.12, [ 0.0,  1.0,  0.35], 10, 0.52, COLORS.white,     COLORS.goldCenter],

          [-0.78,  1.05,  0.22, [-0.55, 0.85, 0.45],  8, 0.48, COLORS.pink,      COLORS.goldCenter],

          [ 0.82,  0.96,  0.18, [ 0.55, 0.82, 0.42], 12, 0.50, COLORS.yellow,    COLORS.brownCenter],

          [-0.38,  0.72,  0.62, [-0.30, 0.65, 0.85],  8, 0.44, COLORS.purple,    COLORS.goldCenter],

          [ 0.45,  0.68,  0.60, [ 0.35, 0.65, 0.82],  9, 0.45, COLORS.lightPink, COLORS.goldCenter],

          [-1.08,  0.55, -0.10, [-0.85, 0.55, 0.15], 10, 0.44, COLORS.orange,    COLORS.brownCenter],

          [ 1.05,  0.52, -0.18, [ 0.85, 0.52, 0.05],  8, 0.44, COLORS.pink,      COLORS.goldCenter],

          [ 0.05,  0.88, -0.58, [ 0.05, 0.85,-0.65],  9, 0.46, COLORS.purple,    COLORS.goldCenter],

          [-0.65,  0.75, -0.52, [-0.55, 0.78,-0.55], 10, 0.45, COLORS.yellow,    COLORS.brownCenter],

          [ 0.68,  0.78, -0.50, [ 0.55, 0.80,-0.52],  8, 0.44, COLORS.orange,    COLORS.goldCenter]

        ];


        for (const [x, y, z, dir, petals, r, col, center] of cosmosSpecs) {

          addCosmos(x, y, z, dir, petals, r, col, center);

          addStem(x, y - 0.1, z, x * 0.15, -1.72, z * 0.15);

        }


        // Tulipes rouges, roses et jaunes

        const tulipSpecs = [

          [ 0.38,  1.22, -0.12, [ 0.25, 1.0,  0.05], 1.05, COLORS.red],

          [-0.45,  1.18, -0.18, [-0.25, 1.0, -0.05], 1.00, COLORS.red],

          [ 0.02,  0.48,  0.78, [ 0.0,  0.55, 0.90], 0.92, COLORS.red],

          [-0.82,  0.32,  0.45, [-0.65, 0.45, 0.65], 0.90, COLORS.yellow],

          [ 0.85,  0.32,  0.42, [ 0.65, 0.45, 0.62], 0.90, COLORS.pink]

        ];


        for (const [x, y, z, dir, scale, col] of tulipSpecs) {

          addTulip(x, y, z, dir, scale, col);

          addStem(x, y, z, x * 0.12, -1.72, z * 0.12);

        }


        // Feuillage vert autour de la couronne

        const leafCount = 14;

        for (let i = 0; i < leafCount; i++) {

          const a = (i / leafCount) * Math.PI * 2 + 0.15;

          const dir = [Math.cos(a) * 0.92, 0.35 + (i % 3) * 0.12, Math.sin(a) * 0.92];

          addLeaf(

            Math.cos(a) * 0.42, 0.22, Math.sin(a) * 0.42,

            dir,

            0.85 + (i % 2) * 0.18,

            0.20,

            i % 2 === 0 ? COLORS.leafGreen : COLORS.darkGreen

          );

        }


        addWrap();

        addRibbon();

      }


      // Particules flottantes autour du bouquet

      const sparkles = Array.from({ length: 28 }, (_, i) => ({

        angle: (i / 28) * Math.PI * 2,

        radius: 1.45 + Math.random() * 0.85,

        yBase: -1.2 + Math.random() * 2.8,

        speed: 0.4 + Math.random() * 0.6,

        phase: Math.random() * Math.PI * 2,

        ch: i % 3 === 0 ? "*" : (i % 3 === 1 ? "+" : "."),

        color: i % 3 === 0 ? COLORS.lightPink : (i % 3 === 1 ? COLORS.yellow : COLORS.white)

      }));


      buildBouquet();


      // ==========================================

      // Caméra & Interactions (Souris + Tactile)

      // ==========================================

      let yaw = 0;

      let pitch = 0.14;

      let zoom = 1.0;


      let velYaw = 0;

      let velPitch = 0;

      const AUTO_SPIN_SPEED = 0.0085;


      const activePointers = new Map();

      let isDragging = false;

      let lastX = 0;

      let lastY = 0;

      let lastPinchDist = 0;

      let lastTapTime = 0;


      canvas.addEventListener("pointerdown", (e) => {

        canvas.setPointerCapture(e.pointerId);

        activePointers.set(e.pointerId, { x: e.clientX, y: e.clientY });


        // Double-tap / double-clic pour réinitialiser la vue

        const now = performance.now();

        if (activePointers.size === 1 && now - lastTapTime < 280) {

          pitch = 0.14;

          zoom = 1.0;

          velYaw = 0;

          velPitch = 0;

        }

        lastTapTime = now;


        if (activePointers.size === 1) {

          isDragging = true;

          lastX = e.clientX;

          lastY = e.clientY;

        } else if (activePointers.size === 2) {

          isDragging = false;

          const pts = [...activePointers.values()];

          lastPinchDist = Math.hypot(pts[0].x - pts[1].x, pts[0].y - pts[1].y);

        }

      });


      canvas.addEventListener("pointermove", (e) => {

        if (!activePointers.has(e.pointerId)) return;

        activePointers.set(e.pointerId, { x: e.clientX, y: e.clientY });


        if (activePointers.size === 1 && isDragging) {

          const dx = e.clientX - lastX;

          const dy = e.clientY - lastY;

          lastX = e.clientX;

          lastY = e.clientY;


          velYaw = dx * 0.009;

          velPitch = dy * 0.007;

          yaw += velYaw;

          pitch = Math.max(-0.95, Math.min(1.05, pitch + velPitch));

        } else if (activePointers.size === 2) {

          const pts = [...activePointers.values()];

          const dist = Math.hypot(pts[0].x - pts[1].x, pts[0].y - pts[1].y);

          if (lastPinchDist > 0) {

            zoom = Math.max(0.55, Math.min(2.1, zoom * (dist / lastPinchDist)));

          }

          lastPinchDist = dist;

        }

      });


      function endPointer(e) {

        activePointers.delete(e.pointerId);

        if (activePointers.size === 1) {

          const [pt] = activePointers.values();

          isDragging = true;

          lastX = pt.x;

          lastY = pt.y;

        } else if (activePointers.size === 0) {

          isDragging = false;

          lastPinchDist = 0;

        }

      }


      canvas.addEventListener("pointerup", endPointer);

      canvas.addEventListener("pointercancel", endPointer);


      canvas.addEventListener("wheel", (e) => {

        e.preventDefault();

        const factor = e.deltaY > 0 ? 0.92 : 1.08;

        zoom = Math.max(0.55, Math.min(2.1, zoom * factor));

      }, { passive: false });


      // ==========================================

      // Grille ASCII & Buffers pré-alloués

      // ==========================================

      let cols = 0;

      let rows = 0;

      let charW = 8;

      let charH = 14;

      let fontSize = 13;

      let zBuf = new Float32Array(0);

      let chBuf = [];

      let colBuf = [];


      function resize() {

        const dpr = Math.min(window.devicePixelRatio || 1, 2);

        const w = window.innerWidth;

        const h = window.innerHeight;


        canvas.width = Math.floor(w * dpr);

        canvas.height = Math.floor(h * dpr);

        ctx.setTransform(dpr, 0, 0, dpr, 0, 0);


        // Taille de police adaptative selon l'écran (mobile, iPad, PC)

        const minDim = Math.min(w, h);

        fontSize = Math.max(9, Math.min(15, Math.floor(minDim / 52)));

        ctx.font = `bold ${fontSize}px "JetBrains Mono", "Fira Code", Consolas, monospace`;

        ctx.textBaseline = "top";


        charW = ctx.measureText("M").width;

        charH = Math.ceil(fontSize * 1.15);


        cols = Math.max(24, Math.floor(w / charW));

        rows = Math.max(20, Math.floor(h / charH));


        const size = cols * rows;

        zBuf = new Float32Array(size);

        chBuf = new Array(size);

        colBuf = new Array(size);

      }


      window.addEventListener("resize", resize);

      resize();


      // ==========================================

      // Boucle de rendu 3D -> ASCII

      // ==========================================

      const lightDir = normalize([0.45, 0.72, 0.85]);


      function render(time) {

        requestAnimationFrame(render);


        // Rotation automatique + inertie fluide après un glisser

        if (!isDragging) {

          yaw += AUTO_SPIN_SPEED + velYaw;

          pitch = Math.max(-0.95, Math.min(1.05, pitch + velPitch));

          velYaw *= 0.92;

          velPitch *= 0.92;

        }


        const size = cols * rows;

        zBuf.fill(-1e9);

        chBuf.fill("");


        const cosY = Math.cos(yaw);

        const sinY = Math.sin(yaw);

        const cosX = Math.cos(pitch);

        const sinX = Math.sin(pitch);


        // Zone réservée au texte en bas

        const usableRows = rows - 4;

        const centerCol = cols * 0.5;

        const centerRow = usableRows * 0.52;


        // Mise à l'échelle automatique (garde le bouquet entier sur portrait et paysage)

        const charAspect = charH / charW;

        const scaleY = Math.min(usableRows * 0.25, (cols / charAspect) * 0.28) * zoom;

        const scaleX = scaleY * charAspect;

        const camDist = 5.4;


        // 1. Projection des points du bouquet

        for (let i = 0; i < points.length; i++) {

          const p = points[i];


          // Rotation Y puis X

          const x1 = p.x * cosY - p.z * sinY;

          const z1 = p.x * sinY + p.z * cosY;

          const y2 = p.y * cosX - z1 * sinX;

          const z2 = p.y * sinX + z1 * cosX;


          const invZ = camDist / (camDist - z2);

          if (invZ <= 0) continue;


          const sx = Math.round(centerCol + x1 * scaleX * invZ);

          const sy = Math.round(centerRow - y2 * scaleY * invZ);


          if (sx < 0 || sx >= cols || sy < 0 || sy >= usableRows) continue;


          const idx = sy * cols + sx;

          if (z2 > zBuf[idx]) {

            zBuf[idx] = z2;


            // Normale pivotée pour l'éclairage 3D

            const nx1 = p.nx * cosY - p.nz * sinY;

            const nz1 = p.nx * sinY + p.nz * cosY;

            const ny2 = p.ny * cosX - nz1 * sinX;

            const nz2 = p.ny * sinX + nz1 * cosX;


            const dot = nx1 * lightDir[0] + ny2 * lightDir[1] + nz2 * lightDir[2];

            const lum = Math.max(0.18, Math.min(1.0, 0.32 + Math.max(0, dot) * 0.72));


            const rampIdx = Math.min(RAMP.length - 1, Math.floor(lum * RAMP.length));

            chBuf[idx] = p.ch || RAMP[rampIdx];


            const shade = 0.48 + lum * 0.58;

            const r = Math.min(255, Math.round(p.r * shade));

            const g = Math.min(255, Math.round(p.g * shade));

            const b = Math.min(255, Math.round(p.b * shade));

            colBuf[idx] = `rgb(${r},${g},${b})`;

          }

        }


        // 2. Particules flottantes

        const tSec = time * 0.001;

        for (let i = 0; i < sparkles.length; i++) {

          const sp = sparkles[i];

          const a = sp.angle + tSec * 0.25 * sp.speed;

          const px = Math.cos(a) * sp.radius;

          const pz = Math.sin(a) * sp.radius;

          const py = sp.yBase + Math.sin(tSec * sp.speed + sp.phase) * 0.22;


          const x1 = px * cosY - pz * sinY;

          const z1 = px * sinY + pz * cosY;

          const y2 = py * cosX - z1 * sinX;

          const z2 = py * sinX + z1 * cosX;


          const invZ = camDist / (camDist - z2);

          const sx = Math.round(centerCol + x1 * scaleX * invZ);

          const sy = Math.round(centerRow - y2 * scaleY * invZ);


          if (sx >= 0 && sx < cols && sy >= 0 && sy < usableRows) {

            const idx = sy * cols + sx;

            if (z2 > zBuf[idx]) {

              const twinkle = 0.45 + 0.55 * Math.sin(tSec * 3.0 + sp.phase);

              if (twinkle > 0.35) {

                zBuf[idx] = z2;

                chBuf[idx] = sp.ch;

                colBuf[idx] = `rgb(${Math.round(sp.color[0]*twinkle)},${Math.round(sp.color[1]*twinkle)},${Math.round(sp.color[2]*twinkle)})`;

              }

            }

          }

        }


        // 3. Dessin sur fond noir pur

        ctx.fillStyle = "#000000";

        ctx.fillRect(0, 0, window.innerWidth, window.innerHeight);


        ctx.font = `bold ${fontSize}px "JetBrains Mono", "Fira Code", Consolas, monospace`;

        const offsetX = (window.innerWidth - cols * charW) * 0.5;

        const offsetY = (window.innerHeight - rows * charH) * 0.5;


        for (let y = 0; y < usableRows; y++) {

          const rowOff = y * cols;

          const py = offsetY + y * charH;

          for (let x = 0; x < cols; x++) {

            const ch = chBuf[rowOff + x];

            if (ch) {

              ctx.fillStyle = colBuf[rowOff + x];

              ctx.fillText(ch, offsetX + x * charW, py);

            }

          }

        }


        // 4. Texte ASCII en bas ("I love you sweetheart <3")

        const msgFontSize = Math.max(12, Math.min(20, Math.floor(window.innerWidth / (MESSAGE.length * 0.72))));

        ctx.font = `bold ${msgFontSize}px "JetBrains Mono", "Fira Code", Consolas, monospace`;

        const pulse = 0.85 + 0.15 * Math.sin(tSec * 2.2);

        const mr = Math.round(255 * pulse);

        const mg = Math.round(125 * pulse);

        const mb = Math.round(180 * pulse);


        ctx.fillStyle = `rgb(${mr}, ${mg}, ${mb})`;

        const textWidth = ctx.measureText(MESSAGE).width;

        const textX = (window.innerWidth - textWidth) * 0.5;

        const textY = window.innerHeight - msgFontSize * 2.2;

        ctx.fillText(MESSAGE, textX, textY);

      }


      requestAnimationFrame(render);

    })();

  </script>
